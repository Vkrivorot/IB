class LinearCongruentialGenerator:
    def __init__(self, seed, a=1664525, c=1013904223, m=2**32):
        self.state = seed
        self.a = a
        self.c = c
        self.m = m

    def next(self):
        # Генерируем следующее значение
        self.state = (self.a * self.state + self.c) % self.m
        return self.state

    def generate_sequence(self, n):
        # Генерируем последовательность из n чисел
        return [self.next() for _ in range(n)]

# Пример использования
if __name__ == "__main__":
    seed = int(input("Введите начальное значение (seed): "))
    lcg = LinearCongruentialGenerator(seed)
    
    n = int(input("Введите количество случайных чисел для генерации: "))
    random_numbers = lcg.generate_sequence(n)
    
    print("Сгенерированные случайные числа:", random_numbers)
