def generate_magic_square(n):
    if n % 2 == 0:
        raise ValueError("Order must be odd.")

    magic_square = [[0] * n for _ in range(n)]
    num = 1
    i, j = 0, n // 2  # Начинаем с первой строки и центрального столбца

    while num <= n * n:
        magic_square[i][j] = num
        num += 1
        new_i, new_j = (i - 1) % n, (j + 1) % n  # Переход вверх и вправо

        if magic_square[new_i][new_j]:  # Если ячейка занята
            i += 1  # Переход вниз
        else:
            i, j = new_i, new_j  # Переход на новую позицию

    return magic_square

def print_magic_square(square):
    for row in square:
        print(" ".join(f"{num:2d}" for num in row))

def encrypt_with_magic_square(text, magic_square):
    encrypted_text = ""
    for char in text:
        if char.isalpha():  # Проверяем, является ли символ буквой
            index = ord(char.upper()) - ord('A')  # Преобразуем букву в индекс (0-25)
            row = index // len(magic_square)
            col = index % len(magic_square)
            encrypted_text += str(magic_square[row][col]) + " "
        else:
            encrypted_text += char  # Оставляем символы, не являющиеся буквами, без изменений
    return encrypted_text.strip()

# Пример использования
order = int(input("Введите порядок магического квадрата (нечетное число): "))
try:
    magic_square = generate_magic_square(order)
    print("Сгенерированный магический квадрат:")
    print_magic_square(magic_square)

    text_to_encrypt = input("Введите текст для шифрования: ")
    encrypted_text = encrypt_with_magic_square(text_to_encrypt, magic_square)
    print("Зашифрованный текст:", encrypted_text)
except ValueError as e:
    print(e)
