import random

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def is_prime(n):
    if n <= 1:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

def next_prime(n):
    while not is_prime(n):
        n += 1
    return n

def power_mod(base, exp, mod):
    result = 1
    base = base % mod
    while exp > 0:
        if (exp % 2) == 1:  # Если exp нечетное
            result = (result * base) % mod
        exp = exp >> 1  # Делим exp на 2
        base = (base * base) % mod
    return result

def elgamal_encrypt(p, g, a, message):
    y = power_mod(g, a, p)  # Открытый ключ
    encrypted_message = []

    for char in message:
        k = random.randint(1, p - 2)  # Случайное число k
        c1 = power_mod(g, k, p)  # c1 = g^k mod p
        key = power_mod(y, k, p)  # Генерация ключа
        c2 = ord(char) ^ key  # Шифрование символа
        encrypted_message.append((c1, c2))

    return y, encrypted_message

def elgamal_decrypt(p, a, encrypted_message):
    decrypted_message = ""

    for c1, c2 in encrypted_message:
        key = power_mod(c1, a, p)  # Генерация ключа для дешифровки
        decrypted_char = chr(c2 ^ key)  # Дешифрование символа
        decrypted_message += decrypted_char

    return decrypted_message

# Пример использования
if __name__ == "__main__":
    p = 11  # Простое число
    g = 3   # Генератор
    a = 6   # Секретный ключ

    message = "HELLO"  # Сообщение для шифрования
    y, encrypted_message = elgamal_encrypt(p, g, a, message)

    print("Открытый ключ y:", y)
    print("Зашифрованное сообщение:", encrypted_message)

    decrypted_message = elgamal_decrypt(p, a, encrypted_message)
    print("Дешифрованное сообщение:", decrypted_message)
