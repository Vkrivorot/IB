def generate_matrix(key):
    key = key.upper().replace('J', 'I')  # Заменяем 'J' на 'I'
    matrix = []
    used_chars = set()

    # Добавляем символы из ключа
    for char in key:
        if char not in used_chars and char.isalpha():
            used_chars.add(char)
            matrix.append(char)

    # Добавляем оставшиеся буквы алфавита
    for char in 'ABCDEFGHIKLMNOPQRSTUVWXYZ':
        if char not in used_chars:
            matrix.append(char)

    return matrix  # Возвращаем одномерный список

def prepare_text(text):
    text = text.upper().replace('J', 'I')
    prepared = ''
    i = 0

    while i < len(text):
        if i + 1 < len(text) and text[i] == text[i + 1]:
            prepared += text[i] + 'X'  # Добавляем 'X' между одинаковыми буквами
            i += 1
        else:
            prepared += text[i]
            i += 1

    if len(prepared) % 2 != 0:
        prepared += 'X'  # Добавляем 'X' в конце, если длина нечетная

    return prepared

def encrypt(text, key):
    matrix = generate_matrix(key)
    prepared_text = prepare_text(text)

    encrypted_text = ''

    for i in range(0, len(prepared_text), 2):
        first_char = prepared_text[i]
        second_char = prepared_text[i + 1]

        # Проверка на наличие символов в матрице
        if first_char not in matrix or second_char not in matrix:
            raise ValueError(f"Characters '{first_char}' or '{second_char}' not found in matrix.")

        row1, col1 = divmod(matrix.index(first_char), 5)
        row2, col2 = divmod(matrix.index(second_char), 5)

        if row1 == row2:  # Одинаковые строки
            encrypted_text += matrix[row1 * 5 + (col1 + 1) % 5]
            encrypted_text += matrix[row2 * 5 + (col2 + 1) % 5]
        elif col1 == col2:  # Одинаковые столбцы
            encrypted_text += matrix[((row1 + 1) % 5) * 5 + col1]
            encrypted_text += matrix[((row2 + 1) % 5) * 5 + col2]
        else:  # Разные строки и столбцы
            encrypted_text += matrix[row1 * 5 + col2]
            encrypted_text += matrix[row2 * 5 + col1]

    return encrypted_text

# Пример использования
key = input("Введите ключ для шифрования: ")
text_to_encrypt = input("Введите текст для шифрования: ")
encrypted_text = encrypt(text_to_encrypt, key)
print("Зашифрованный текст:", encrypted_text)
